# Core dependencies for the Document Intelligence Agent.

# API + serving
fastapi
uvicorn            # ASGI server (also serves the web UI at /ui)
python-multipart   # FastAPI multipart parsing (file uploads)

# PDF text extraction
pymupdf            # fast per-page extraction (import as `pymupdf`)

# embeddings + vector store
sentence-transformers   # all-MiniLM-L6-v2 (CPU)
chromadb                # persistent vector database

# retrieval
rank-bm25          # BM25 for hybrid (dense + keyword) retrieval

# Streamlit UI (the web chat UI at /ui needs no extra deps)
streamlit
pandas             # evaluation dashboard table/chart

# --- testing ---
pytest
httpx              # FastAPI TestClient + the vision/LLM HTTP calls
